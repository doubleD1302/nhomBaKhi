📜 Hướng dẫn chuẩn bị Database trên Supabase (Chỉ cần 1 người chạy 1 lần)
  1. Đăng ký/đăng nhập supabase.com và tạo một New Project miễn phí.
  2. Vào mục SQL Editor → Dán đoạn SQL bên dưới → Bấm Run:
CREATE TABLE IF NOT EXISTS bang_cong_viec (
    task_id VARCHAR(10) PRIMARY KEY,
    status VARCHAR(20) DEFAULT 'todo',
    progress_percent INT DEFAULT 0,
    actual_hours NUMERIC(5,2) DEFAULT 0,
    pr_link TEXT DEFAULT '',
    notes TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    updated_by VARCHAR(50) DEFAULT ''
);
ALTER TABLE bang_cong_viec ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Cho phep nhom doc ghi tien do" ON bang_cong_viec;
CREATE POLICY "Cho phep nhom doc ghi tien do" ON bang_cong_viec FOR ALL USING (true) WITH CHECK (true);
